-- Настройка gethui / Системы защиты
do
    local function d(aa)
        local ab = type(aa) == 'table' and rawget(aa, 'gethui')
        if type(ab) ~= 'function' or rawget(aa, 'zamlhui') then return end
        local ac, ad = pcall(ab)
        if not (ac and (typeof(ad) == 'Instance' and ad:IsA'LayerCollector')) then return end
        rawset(aa, 'zamlhui', true)
        rawset(aa, 'gethui', function()
            local ae = ab()
            if typeof(ae) == 'Instance' and ae:IsA'LayerCollector' then
                return ae.Parent or game:GetService'CoreGui'
            end
            return ae
        end)
    end
    pcall(d, getfenv and getfenv(1))
    pcall(d, getgenv and getgenv())
end

local aa = { unloaded = false, conns = {}, cleanups = {} }
do
    local ab = getgenv and getgenv() or _G
    if type(ab.nashot) == 'table' and type(ab.nashot.unload) == 'function' then pcall(ab.nashot.unload) end
    if type(ab.__nashotUnload) == 'function' then pcall(ab.__nashotUnload) end

    function aa.connect(ac, ad)
        local ae, af = ac:Connect(ad), aa.conns
        af[#af + 1] = ae
        if aa.unloaded then ae:Disconnect() end
        return ae
    end
    function aa.onUnload(ac) aa.cleanups[#aa.cleanups + 1] = ac end
    function aa.unload()
        if aa.unloaded then return end
        aa.unloaded = true
        for ac, ad in ipairs(aa.conns) do pcall(function() ad:Disconnect() end) end
        for ac = #aa.cleanups, 1, -1 do pcall(aa.cleanups[ac]) end
        if type(ab.nashot) == 'table' and ab.nashot.unload == aa.unload then ab.nashot = nil end
    end
    ab.nashot = { unload = aa.unload }
end

-- Загрузка UI библиотеки WindUI
local ac
do
    local ad
    pcall(function()
        if not isfolder'nashot' then makefolder'nashot' end
        local ae, af = pcall(game.HttpGet, game, 'https://raw.githubusercontent.com/amazson/zaml/main/windui.lua')
        if ae and (type(af) == 'string' and #af > 10000) then
            writefile('nashot/windui.lua', af)
            ad = af
        end
    end)
    for ae, af in ipairs{'nashot/windui.lua', 'windui.lua'} do
        if not ad and (isfile and isfile(af)) then
            local ag, ah = pcall(readfile, af)
            if ag and (type(ah) == 'string' and #ah > 10000) then ad = ah end
        end
    end
    if not ad then error'[nashot] windui.lua not found' end
    local ae, af = loadstring(ad, '=windui')
    if not ae then error('[nashot] windui failed: ' .. tostring(af)) end
    local ag, ah = pcall(ae)
    ac = ah
end

-- [Функционал Легит: Silent Aim, No Recoil, No Spread]
local at = workspace.CurrentCamera
local bg = nil

local function getTarget()
    local maxDist = aC.fov
    local target = nil
    local center = at.ViewportSize / 2

    for _, p in ipairs(u:GetPlayers()) do
        if p ~= _ and p.Character and p.Character:FindFirstChild(aC.aimPart) then
            if not aC.teamCheck or p.Team ~= _.Team then
                local part = p.Character[aC.aimPart]
                local pos, onScreen = at:WorldToViewportPoint(part.Position)
                if onScreen then
                    local screenPos = Vector2.new(pos.X, pos.Y)
                    local dist = (screenPos - center).Magnitude
                    if dist < maxDist then
                        maxDist = dist
                        target = part
                    end
                end
            end
        end
    end
    return target
end

-- Хук выстрела для Silent Aim / No Recoil / No Spread
pcall(function()
    local remotes = require(L.Database.Security.Remotes)
    local bullet = require(L.Components.Weapon.Classes.Bullet)
    
    -- Блокировка разброса
    local oldSpread
    oldSpread = hookfunction(bullet.getTrueSpread, function(self, ...)
        if aC.noSpread then return 0 end
        return oldSpread(self, ...)
    end)

    -- Блокировка отдачи
    local cameraController = require(L.Controllers.CameraController)
    local oldRecoil
    oldRecoil = hookfunction(game:GetService("ReplicatedStorage").Classes.WeaponComponent.Classes.Viewmodel.setWeaponRecoil, function(self, val)
        if aC.noRecoil then return oldRecoil(self, Vector3.zero) end
        return oldRecoil(self, val)
    end)
end)
