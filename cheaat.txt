-- Скрипт плавного перетаскивания главного меню
local dragging, dragInput, dragStart, startPos
MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)
MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Функция создания кнопок

    btn.Text = text
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)

    btn.BorderSizePixel = 0
    
    local btnCorner = Instance.new("UICorner")

end

createButton(EspButton, "ESP: OFF", 45)
createButton(AimButton, "SILENT AIM: OFF", 95)

-- --- 3. РАБОЧИЙ ESP ---
local espEnabled = false
local function toggleEsp()
    espEnabled = not espEnabled
    EspButton.Text = espEnabled and "ESP: ON" or "ESP: OFF"
    EspButton.BackgroundColor3 = espEnabled and Color3.fromRGB(0, 180, 100) or Color3.fromRGB(45, 45, 45)
    
    if not espEnabled then
        for _, p in pairs(Players:GetPlayers()) do
            if p.Character and p.Character:FindFirstChild("DeltaESP") then
                p.Character.DeltaESP:Destroy()
            end
        end
    end
end

RunService.RenderStepped:Connect(function()
    if espEnabled then
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then
                local isEnemy = true
                if p.Team and LocalPlayer.Team and p.Team == LocalPlayer.Team then
                    isEnemy = false
                end
                
                if isEnemy then
                    if not p.Character:FindFirstChild("DeltaESP") then
                        local hl = Instance.new("Highlight")
                        hl.Name = "DeltaESP"
                        hl.Parent = p.Character
                        hl.FillColor = Color3.fromRGB(255, 50, 50)
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.4
                        hl.OutlineTransparency = 0
                    end
                else
                    if p.Character:FindFirstChild("DeltaESP") then
                        p.Character.DeltaESP:Destroy()
                    end
                end
            end
        end
    end
end)

-- --- 4. ВИЗУАЛИЗАЦИЯ КРУГА FOV ---
local FOV_Circle = Drawing.new("Circle")
FOV_Circle.Visible = false
FOV_Circle.Transparency = 0.7
FOV_Circle.Color = Color3.fromRGB(255, 255, 255)
FOV_Circle.Thickness = 1
FOV_Circle.Filled = false
FOV_Circle.Radius = FOV_Radius

RunService.RenderStepped:Connect(function()
    if SilentAimEnabled and ShowFOV then
        FOV_Circle.Visible = true
        FOV_Circle.Radius = FOV_Radius
        FOV_Circle.Position = UserInputService:GetMouseLocation()
    else
        FOV_Circle.Visible = false
    end
end)

-- --- 5. SILENT AIM ---
local function getClosestEnemyInFOV()
    local closestTarget = nil
    local shortestDistance = FOV_Radius
    local mousePos = UserInputService:GetMouseLocation()
    
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
            local isEnemy = true
            if p.Team and LocalPlayer.Team and p.Team == LocalPlayer.Team then
                isEnemy = false
            end
            
            if isEnemy then
                local head = p.Character.Head
                local screenPos, onScreen = Camera:WorldToScreenPoint(head.Position)
                
                if onScreen then
                    local targetScreenPos = Vector2.new(screenPos.X, screenPos.Y)
                    local distance = (mousePos - targetScreenPos).Magnitude
                    
                    if distance < shortestDistance then
                        shortestDistance = distance
                        closestTarget = head
                    end
                end
            end
        end
    end
    return closestTarget
end

local function toggleAim()
    SilentAimEnabled = not SilentAimEnabled
    AimButton.Text = SilentAimEnabled and "SILENT AIM: ON" or "SILENT AIM: OFF"
    AimButton.BackgroundColor3 = SilentAimEnabled and Color3.fromRGB(0, 180, 100) or Color3.fromRGB(45, 45, 45)
end

local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if SilentAimEnabled and (method == "FindPartOnRay" or method == "FindPartOnRayWithIgnoreList" or method == "FindPartOnRayWithWhitelist") then
        local targetHead = getClosestEnemyInFOV()
        if targetHead and args[1] then
            local origin = args[1].Origin
            local direction = (targetHead.Position - origin).Unit * 1000
            args[1] = Ray.new(origin, direction)
            return oldNamecall(self, unpack(args))
        end
    end
    
    return oldNamecall(self, ...)
end)

-- Обработка нажатий кнопок
EspButton.MouseButton1Click:Connect(toggleEsp)
AimButton.MouseButton1Click:Connect(toggleAim)
